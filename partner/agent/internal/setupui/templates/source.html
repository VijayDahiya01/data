How to create a read-only login

Replace shop, customers and the password with your own.

PostgreSQL

CREATE ROLE oolix_reader LOGIN PASSWORD 'choose-a-strong-password';
GRANT CONNECT ON DATABASE shop TO oolix_reader;
GRANT USAGE ON SCHEMA public TO oolix_reader;
GRANT SELECT ON public.customers TO oolix_reader;

MySQL / MariaDB

CREATE USER 'oolix_reader'@'%' IDENTIFIED BY 'choose-a-strong-password';
GRANT SELECT ON shop.customers TO 'oolix_reader'@'%';

Microsoft SQL Server

CREATE LOGIN oolix_reader WITH PASSWORD = 'choose-a-strong-password';
USE shop;
CREATE USER oolix_reader FOR LOGIN oolix_reader;
GRANT SELECT ON dbo.customers TO oolix_reader;

MongoDB

use admin
db.createUser({ user: "oolix_reader", pwd: "choose-a-strong-password",
                roles: [{ role: "read", db: "shop" }] })
